│  │ @PostMapping("/inventory-assistant")                                     │  │
│  │ @PreAuthorize("isAuthenticated()")                                       │  │
│  │ public InventoryAssistantResponse inventoryAssistant(                   │  │
│  │   @RequestBody InventoryAssistantRequest request) {                     │  │
│  │                                                                          │  │
│  │   // 1. Extraer rol del usuario                                         │  │
│  │   RoleContextInfo roleContext = extractRoleContext()                    │  │
│  │                                                                          │  │
│  │   // 2. Validar autenticación                                            │  │
│  │   if (!roleContext.isAuthenticated())                                    │  │
│  │     throw new IllegalStateException("No autenticado")                    │  │
│  │                                                                          │  │
│  │   // 3. Validar acceso al chatbot                                        │  │
│  │   if (!roleContext.hasAccessToChatbot())                                 │  │
│  │     throw new IllegalStateException("Rol sin acceso")                    │  │
│  │                                                                          │  │
│  │   // 4. Pasar a servicio                                                │  │
│  │   return inventoryAssistantService                                       │  │
│  │     .answerInventoryQuestion(request, roleContext)                      │  │
│  │ }                                                                        │  │
│  │                                                                          │  │
│  │ private RoleContextInfo extractRoleContext() {                          │  │
│  │   Authentication auth = SecurityContextHolder.getContext()              │  │
│  │                         .getAuthentication()                            │  │
│  │   String username = auth.getName()   // "admin"                        │  │
│  │   String userRole = auth.getAuthorities()                              │  │
│  │     .stream()                                                            │  │
│  │     .filter(a -> a.startsWith("ROLE_"))                                │  │
│  │     .map(a -> a.substring(5))       // "ROLE_ADMIN" → "ADMIN"         │  │
│  │     .findFirst()                                                         │  │
│  │     .orElse("RECEPCION")                                                │  │
│  │                                                                          │  │
│  │   return new RoleContextInfo(null, username, userRole, ...)            │  │
│  │ }                                                                        │  │
│  │                                                                          │  │

│  │ public InventoryAssistantResponse answerInventoryQuestion(             │  │
│  │   InventoryAssistantRequest request,                                   │  │
│  │   RoleContextInfo roleContext) {                                        │  │
│  │                                                                          │  │
│  │   // 1. Cargar contexto COMPLETO                                        │  │
│  │   InventorySnapshot fullSnapshot = loadContext()                       │  │
│  │   // → items (142), lowStock (15), users (24), rooms (80),             │  │
│  │   //   movements, alerts, providers, etc.                              │  │
│  │                                                                          │  │
│  │   // 2. FILTRAR contexto según rol                                     │  │
│  │   FilteredContextSnapshot filtered =                                   │  │
│  │     filterContextByRole(fullSnapshot, roleContext.role())              │  │
│  │                                                                          │  │
│  │   // 3. Obtener instrucciones adaptadas                                │  │
│  │   String instructions =                                                 │  │
│  │     promptBuilder.getAiInstructionsForRole(role)                       │  │
│  │   // → "Eres asistente de bodega. Tienes acceso a: items,            │  │
│  │   //    lowStock, alerts. NO tienes: habitaciones, usuarios..."        │  │
│  │                                                                          │  │
│  │   // 4. Construir prompt con contexto filtrado                         │  │
│  │   String prompt = buildPromptWithFilteredContext(                      │  │
│  │     question, filtered)                                                 │  │
│  │                                                                          │  │
│  │   // 5. Llamar a Gemini                                                │  │
│  │   String answer = geminiClient                                          │  │
│  │     .generateInventoryAnswer(instructions, prompt)                     │  │
│  │                                                                          │  │
│  │   // 6. Retornar respuesta                                             │  │
│  │   return new InventoryAssistantResponse(answer, contextSource)         │  │
│  │ }                                                                        │  │
